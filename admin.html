// ./js/modules/app-admin.js document.addEventListener('DOMContentLoaded', () => { const API_URL = 'https://huby.onrender.com'; const defaultCoords = [-7.7735, -42.2472]; // Paes Landim - PI // 1. Inicializar Mapa Administrativo const mapElement = document.getElementById('mapaAdmin'); let map = null; if (mapElement) { map = L.map('mapaAdmin', { zoomControl: false, attributionControl: false }).setView(defaultCoords, 15); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19 }).addTo(map); // Marcador simulado da frota ativa em campo L.circleMarker(defaultCoords, { radius: 10, color: '#10B981', fillColor: '#10B981', fillOpacity: 1, weight: 3 }).addTo(map).bindPopup('Tarcyo Carvalho
Status: Online (HUBy Carro)'); setTimeout(() => map.invalidateSize(), 300); window.addEventListener('resize', () => map.invalidateSize()); } // 2. Sincronizar Fila de Pedidos Ativos com a API e localStorage async function carregarFilaAdmin() { const listaContainer = document.getElementById('listaPedidosAdmin'); if (!listaContainer) return; let pedidos = []; // Tenta buscar no Render try { const resp = await fetch(`${API_URL}/api/pedidos`); if (resp.ok) { pedidos = await resp.json(); } } catch (e) { console.warn('API Admin offline, a carregar dados locais.'); } // Fallback local se a API estiver vazia if (!pedidos || pedidos.length === 0) { pedidos = JSON.parse(localStorage.getItem('huby_pedidos_fila') || '[]'); } if (pedidos.length === 0) { listaContainer.innerHTML = `
Nenhum pedido ou frete ativo no momento.
`; return; } listaContainer.innerHTML = pedidos.map(p => { const isCargo = p.tipo === 'cargo'; const isFlash = p.tipo === 'flash'; const iconeClasse = isCargo ? 'cargo' : (isFlash ? 'flash' : 'emerald'); const emoji = isCargo ? '🚚' : (isFlash ? '📦' : '🚗'); return `
${emoji}
#${p.id} • ${p.subTipoLabel || 'HUBy Viagem'} Origem: ${p.origem} Destino: ${p.destino} (${p.distancia || '1.0 km'})
${p.status || 'Pendente'} ${p.preco}
`; }).join(''); } // Atualiza a cada 5 segundos setInterval(carregarFilaAdmin, 5000); carregarFilaAdmin(); });
